Simulation

Here I aim to look at a simulation with similar strcuture to the pancreas dataset. I simulate the expressions for eight cell types and four donors, with gaussian noise:

\[ X = L_{\mathrm{type}} F_{\mathrm{type}}^\top + L_{\mathrm{donor}} F_{\mathrm{donor}}^\top + E. \]

Every cell type contains all four donors in equal proportions. Donor effects are shared across cell types. Both loading matrices contain centered indicators.

set.seed(1)

I scale the data to give cell-type variance 1 and donor variance 0.25.

L_type <- scale(model.matrix(~ celltype - 1, sample_info), scale = FALSE)

F_type <- matrix(rnorm(p * ncol(L_type)), nrow = p)

E <- matrix(rnorm(n * p, sd = noise_sd), nrow = n)
X <- X_type + X_donor + E

Methods compared

I compare the following rank-1 methods: - Log-Phi with \(\alpha=1\) and \(2\) - FastICA with an intercept - First-order EB-proj with normalized Bernoulli, and centered Exponential. - Goldenplus EB-proj with normalized Bernoulli, and centered Exponential, bith use ICA-type hessian approximation.

I look at 100 random starts and visualize their solutions.

The normalized Bernoulli is given by

\[ \mathbb{P}[z = \sqrt{(1-p)/p}] = p \quad \mathbb{P}[z = -\sqrt{p/(1-p)}] = 1-p, \]

where centered Exponential is defined as

\[ Z = T-1,\quad T\sim \text{Exp}(1). \]

Both have mean \(0\) and variance \(1\). Note that EB-proj with centered Exp is not really an EB method, as the prior is fixed.

c_golden <- 2 / (1 + sqrt(5))

# Scale posterior moments and density; fitted_g stores the unit prior's shape.
warm_prior <- function(x, s, g_init = NULL, fix_g = FALSE, output = NULL) {
  fit <- ebnm_normalized_bernoulli(x / c_golden, s / c_golden,
    g_init = g_init, fix_g = fix_g, output = output)
  fit$posterior$mean <- c_golden * fit$posterior$mean
  fit$posterior$sd <- c_golden * fit$posterior$sd
  fit$posterior$second_moment <- c_golden^2 * fit$posterior$second_moment
  fit$log_likelihood <- as.numeric(fit$log_likelihood) - length(x) * log(c_golden)
  fit
}

fit_goldplus <- function(w) {
  warm <- eb_proj_newton(U, ebnm_fn = warm_prior, w_init = w, g_init = 0.5,
                        sigma2 = c_golden, fix_tau = TRUE, maxiter = maxiter)
  eb_proj_newton(U, ebnm_fn = ebnm_normalized_bernoulli, w_init = warm$w,
                g_init = warm$fitted_g, tau = warm$tau, maxiter = maxiter)
}

P <- tcrossprod(U)
first_order_results <- list()
for (prior_name in names(priors)) {
  prior <- priors[[prior_name]]
  exponential <- prior_name == "Exponential"
  method <- paste0("EBproj-1st-order-", if (exponential) "Exp" else "Bern")
  L <- matrix(0, n, n_starts)
  objective <- numeric(n_starts)
  status <- character(n_starts)
  for (j in seq_len(n_starts)) {
    x_init <- as.vector(t(X1) %*% W_start[, j])
    s_init <- sqrt(n / (n - 2 * n_comp))
    initial <- if (exponential) prior(x_init, s_init) else
      prior(x_init, s_init, g_init = 0.5, fix_g = TRUE)
    fit <- eb_proj_rank1_update(P, initial$posterior$mean, ebnm_fn = prior,
      v2_init = sum(initial$posterior$second_moment),
      k = n_comp, b = n_comp / 2, maxiter = maxiter)
    direction <- as.vector(crossprod(U, fit$vbar))
    direction_norm <- sqrt(sum(direction^2))
    status[j] <- if (fit$niter < maxiter) "converged" else "maximum iterations reached"
    if (fit$v2 == 0) {
      status[j] <- "null solution"
    } else if (direction_norm <= .Machine$double.eps * max(1, sqrt(fit$v2))) {
      status[j] <- "zero projected posterior mean"
    } else {
      L[, j] <- sqrt(n) * U %*% (direction / direction_norm)
    }
    objective[j] <- fit$objective
  }
  first_order_results[[method]] <- list(loadings = L, objective = objective, status = status)
}

Grouped loading plots

Here I group solutions using the method in Matthew’s pancreas note. For each cluster, the first fit is shown, and the plots are ordered by decreasing objective. n counts starts; obj is the representative fit’s objective. Each figure holds up to 15 subplots in three columns, with fixed subplot sizes.

Cells are ordered by cell type, then donor. Each set of components is colored by cell type and by donor.

First-order EB-proj panels plot \(L=\sqrt{n}\,Uw\), where \(w=U^\top m/\|U^\top m\|\) and \(m\) is the posterior mean before projection.

plot_solutions <- function(result, method, start_labels = NULL) {
  L <- result$loadings
  keep <- which(apply(L, 2, sd) > 1e-8 & is.finite(result$objective))
  if (!is.null(result$status)) {
    keep <- keep[result$status[keep] %in% c("converged", "maximum iterations reached")]
  }
  if (!length(keep)) return(invisible(NULL))
  clusters <- if (!is.null(start_labels)) seq_along(keep) else if (length(keep) == 1) 1 else cutree(
    hclust(as.dist(1 - pmin(abs(cor(L[, keep, drop = FALSE])), 1)),
           method = "complete"), h = 0.1)
  groups <- split(keep, clusters)
  representatives <- vapply(groups, function(idx) idx[1], 1L)
  o <- order(result$objective[representatives], decreasing = TRUE)
  representatives <- representatives[o]
  labels <- sprintf("n: %d  obj: %.3f", lengths(groups)[o],
                    result$objective[representatives])
  if (!is.null(start_labels)) labels <- sprintf("%s  obj: %.3f",
    start_labels[representatives], result$objective[representatives])
  cell_order <- order(sample_info$celltype, sample_info$donor)
  pages <- split(seq_along(representatives), ceiling(seq_along(representatives) / 15))
  for (page in seq_along(pages)) {
    idx <- pages[[page]]
    df <- data.frame(
      cell = rep(seq_len(n), length(idx)),
      loading = as.vector(L[cell_order, representatives[idx], drop = FALSE]),
      panel = factor(rep(representatives[idx], each = n), levels = representatives[idx]),
      celltype = rep(sample_info$celltype[cell_order], length(idx)),
      donor = rep(sample_info$donor[cell_order], length(idx)))
    for (color_by in c("celltype", "donor")) {
      p <- ggplot(df, aes(cell, loading, color = .data[[color_by]])) +
        geom_point(size = 0.5, alpha = 0.7) +
        geom_hline(yintercept = 0, linetype = 2, linewidth = 0.3) +
        facet_wrap(~ panel, ncol = 3, scales = "free_y",
                   labeller = labeller(panel = setNames(labels, representatives))) +
        scale_color_manual(values = hcl.colors(nlevels(df[[color_by]]), "Dark 3")) +
        labs(x = NULL, y = "Loading", color = color_by,
             title = paste(method, "-", color_by),
             subtitle = paste("Page", page, "of", length(pages))) +
        theme_bw(base_size = 16) +
        theme(axis.text.x = element_blank(), axis.ticks.x = element_blank(),
              legend.position = "bottom", strip.text = element_text(size = 14)) +
        guides(color = guide_legend(override.aes = list(size = 3, alpha = 1)))
      grid::grid.newpage()
      g <- ggplotGrob(p)
      panels <- grepl("^panel", g$layout$name)
      g$widths[unique(g$layout$l[panels])] <- grid::unit(3.1, "in")
      g$heights[unique(g$layout$t[panels])] <- grid::unit(1.15, "in")
      g$vp <- grid::viewport(x = 0, y = 1, just = c("left", "top"),
                            width = sum(g$widths), height = sum(g$heights))
      grid::grid.draw(g)
    }
  }
}
